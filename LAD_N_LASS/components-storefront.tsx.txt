"use client"

import { useMemo, useState } from "react"
import Image from "next/image"
import { ArrowDown, ArrowRight, ArrowUpRight, Check, ChevronDown, Heart, Instagram, Minus, Plus, Search, ShoppingBag, X } from "lucide-react"

type Product = { id: number; name: string; category: string; price: number; image: string; tag?: string; description: string }
type CartItem = { product: Product; quantity: number }

const products: Product[] = [
  { id: 1, name: "The Sunday Set", category: "Clothing", price: 6800, image: "/images/product-set2.png", tag: "BESTSELLER", description: "A little ease, a lot of intention. Our signature two-piece set is cut for a relaxed fit and made to move with you." },
  { id: 2, name: "Malaika Midi", category: "Dresses", price: 7400, image: "/images/product-dress.png", tag: "JUST IN", description: "Meet the dress you will reach for on repeat. A softly draped silhouette, a flattering waist, and thoughtful details." },
  { id: 3, name: "Woven Story Jacket", category: "Clothing", price: 8900, image: "/images/product-set.png", description: "Handwoven texture meets an easy everyday shape. Each piece celebrates the beauty of craft and the hands behind it." },
  { id: 4, name: "The Nia Crossbody", category: "Accessories", price: 4200, image: "/images/product-bag.png", tag: "MADE BY HAND", description: "A forever kind of bag. Carefully woven raffia, warm leather details, and just enough room for your everyday essentials." },
]
const money = (value: number) => `KSh ${value.toLocaleString("en-KE")}`

export default function Storefront() {
  const [category, setCategory] = useState("All pieces")
  const [cart, setCart] = useState<CartItem[]>([])
  const [drawer, setDrawer] = useState(false)
  const [selected, setSelected] = useState<Product | null>(null)
  const [checkout, setCheckout] = useState(false)
  const [orderPlaced, setOrderPlaced] = useState(false)
  const [searchOpen, setSearchOpen] = useState(false)
  const [query, setQuery] = useState("")
  const [email, setEmail] = useState("")
  const cartCount = cart.reduce((sum, item) => sum + item.quantity, 0)
  const subtotal = cart.reduce((sum, item) => sum + item.product.price * item.quantity, 0)
  const visibleProducts = useMemo(() => products.filter((item) => (category === "All pieces" || item.category === category) && item.name.toLowerCase().includes(query.toLowerCase())), [category, query])

  function addToCart(product: Product) {
    setCart((current) => current.some((item) => item.product.id === product.id) ? current.map((item) => item.product.id === product.id ? { ...item, quantity: item.quantity + 1 } : item) : [...current, { product, quantity: 1 }])
    setSelected(null)
    setDrawer(true)
  }
  function updateQuantity(id: number, delta: number) {
    setCart((current) => current.map((item) => item.product.id === id ? { ...item, quantity: item.quantity + delta } : item).filter((item) => item.quantity > 0))
  }

  return (
    <main className="storefront">
      <div className="announcement">A little something for your everyday <span>—</span> complimentary delivery on orders over KSh 10,000</div>
      <header className="site-header">
        <button className="mobile-menu" aria-label="Open menu" onClick={() => document.getElementById("shop")?.scrollIntoView({ behavior: "smooth" })}><span /><span /></button>
        <nav className="main-nav" aria-label="Main navigation"><a href="#shop">Shop</a><a href="#story">Our story</a><a href="#journal">The journal</a></nav>
        <a className="wordmark" href="#top" aria-label="Lad and Lass home">lad <span>&</span> lass<span className="wordmark-dot">.</span></a>
        <div className="header-actions">
          <button className="icon-button search-toggle" aria-label="Search products" onClick={() => setSearchOpen((open) => !open)}><Search size={19} strokeWidth={1.5} /></button>
          <button className="bag-button" aria-label={`Shopping bag, ${cartCount} items`} onClick={() => { setDrawer(true); setCheckout(false); setOrderPlaced(false) }}><span>Bag</span><span className="bag-count">{cartCount}</span></button>
        </div>
      </header>
      <section className="hero" id="top">
        <div className="hero-copy">
          <p className="eyebrow"><span className="eyebrow-line" /> THE EVERYDAY, ELEVATED</p>
          <h1>Made for<br />the way <em>you</em><br />move.</h1>
          <p className="hero-description">Thoughtful pieces, made with feeling. For all your ordinary days and everything that makes them yours.</p>
          <a className="dark-button" href="#shop">Find your thing <ArrowRight size={16} /></a>
          <div className="hero-note"><span className="note-mark">✳</span><span>Designed with care.<br />Made to be lived in.</span></div>
        </div>
        <div className="hero-visual"><Image src="/images/hero-fashion.png" alt="Model wearing a warm-toned Lad & Lass outfit" fill priority className="hero-image" sizes="(max-width: 760px) 100vw, 58vw" /><div className="image-caption"><span>THE NEW SEASON</span><span>01 / 04</span></div><div className="hero-sticker">a softer<br />kind of<br /><em>statement</em></div></div>
        <div className="hero-bottom"><span>NAIROBI, KENYA</span><span>01 — 04</span><a href="#shop">SCROLL TO DISCOVER <ArrowDown size={13} /></a></div>
      </section>
      <section className="ticker" aria-label="Our values"><div>GOOD THINGS, MADE SLOW <span>✳</span> WEAR IT YOUR WAY <span>✳</span> A LITTLE MORE YOU <span>✳</span> GOOD THINGS, MADE SLOW <span>✳</span> WEAR IT YOUR WAY <span>✳</span></div></section>
      <section className="shop-section" id="shop">
        <div className="section-heading"><div><p className="eyebrow">A GOOD PLACE TO START</p><h2>Pieces to <em>keep.</em></h2></div><a className="text-link" href="#shop">Explore everything <ArrowUpRight size={16} /></a></div>
        <div className="shop-toolbar"><div className="category-tabs" role="group" aria-label="Filter products">{["All pieces", "Clothing", "Dresses", "Accessories"].map((item) => <button key={item} className={category === item ? "category active" : "category"} onClick={() => setCategory(item)}>{item}</button>)}</div>{searchOpen && <label className="search-field"><span className="sr-only">Search by product name</span><Search size={16} /><input autoFocus value={query} onChange={(event) => setQuery(event.target.value)} placeholder="Find your thing…" /><button aria-label="Close search" onClick={() => { setSearchOpen(false); setQuery("") }}><X size={15} /></button></label>}<span className="product-count">0{visibleProducts.length} pieces</span></div>
        <div className="product-grid">{visibleProducts.map((product, index) => <article className="product-card" key={product.id}><button className="product-image-button" aria-label={`View ${product.name}`} onClick={() => setSelected(product)}><span className="product-image-wrap"><Image src={product.image} alt={product.name} fill sizes="(max-width: 640px) 50vw, 25vw" className="product-image" />{product.tag && <span className="product-tag">{product.tag}</span>}</span><span className="quick-view">Quick view <ArrowUpRight size={14} /></span></button><div className="product-meta"><div><button className="product-name" onClick={() => setSelected(product)}>{product.name}</button><p>{product.category} · Made with care</p></div><span className="product-price">{money(product.price)}</span></div><button className="add-link" onClick={() => addToCart(product)}>Add to bag <Plus size={14} /></button></article>)}</div>
        {visibleProducts.length === 0 && <p className="no-results">No pieces found. Try another search.</p>}
        <div className="shop-footnote"><span>01 — 04</span><p>Thoughtful things for a wardrobe that feels like you.</p><span>MADE IN KENYA</span></div>
      </section>
      <section className="story-section" id="story"><div className="story-image"><Image src="/images/hero-fashion.png" alt="Lad & Lass celebrates thoughtful personal style" fill sizes="50vw" className="story-photo" /></div><div className="story-copy"><p className="eyebrow">A NOTE FROM US</p><h2>Clothes with<br />a little <em>more</em><br />feeling.</h2><p>We believe getting dressed should feel like coming home to yourself. Lad & Lass is a love letter to the little things: the first coffee, the long way home, the outfit that just gets you.</p><a href="https://www.instagram.com/lad_n_lass_cloth" target="_blank" rel="noreferrer" className="text-link">A little more about us <ArrowUpRight size={16} /></a><span className="story-scribble">with love, always</span></div></section>
      <section className="newsletter" id="journal"><div><p className="eyebrow">LETTERS FROM L&L</p><h2>A good thing<br />in your <em>inbox.</em></h2></div><form className="newsletter-form" onSubmit={(event) => { event.preventDefault(); setEmail("subscribed") }}><label htmlFor="newsletter-email">Occasional notes, new things, and little joys.</label><div className="email-entry"><input id="newsletter-email" type="email" required placeholder="Your email address" value={email === "subscribed" ? "" : email} onChange={(event) => setEmail(event.target.value)} /><button aria-label="Subscribe to newsletter">{email === "subscribed" ? <Check size={18} /> : <ArrowRight size={18} />}</button></div>{email === "subscribed" ? <span className="subscribe-success">You’re on the list. Talk soon.</span> : <span className="privacy-note">No noise, just the good stuff.</span>}</form></section>
      <footer className="site-footer"><a className="footer-logo" href="#top">lad <span>&</span> lass<span className="wordmark-dot">.</span></a><span className="footer-note">Made with care in Nairobi, Kenya.</span><div className="footer-links"><a href="https://www.instagram.com/lad_n_lass_cloth" target="_blank" rel="noreferrer" aria-label="Visit Lad and Lass on Instagram"><Instagram size={17} /> Instagram</a><a href="#shop">Shop all</a><a href="mailto:hello@ladandlass.co.ke">Say hello <ArrowUpRight size={13} /></a></div><span className="copyright">© 2025 LAD & LASS</span></footer>
      {(drawer || selected) && <div className="overlay" onMouseDown={(event) => { if (event.target === event.currentTarget) { setDrawer(false); setSelected(null) } }}>
        {selected && <section className="quick-modal" role="dialog" aria-modal="true" aria-labelledby="quick-title"><button className="close-button" aria-label="Close product details" onClick={() => setSelected(null)}><X /></button><div className="quick-modal-image"><Image src={selected.image} alt={selected.name} fill sizes="45vw" className="product-image" /></div><div className="quick-modal-copy"><p className="eyebrow">{selected.category} · MADE WITH CARE</p><h2 id="quick-title">{selected.name}</h2><p className="quick-price">{money(selected.price)}</p><p className="quick-description">{selected.description}</p><div className="size-select"><span>One thoughtful size</span><ChevronDown size={15} /></div><button className="dark-button full-button" onClick={() => addToCart(selected)}>Add to bag <ArrowRight size={16} /></button><p className="shipping-note">Complimentary delivery over KSh 10,000</p></div></section>}
        {drawer && <aside className="cart-drawer" role="dialog" aria-modal="true" aria-labelledby="cart-title"><div className="drawer-header"><div><p className="eyebrow">YOUR LITTLE FINDS</p><h2 id="cart-title">{orderPlaced ? "It’s a start." : checkout ? "Your details." : "Your bag."}</h2></div><button className="close-button" aria-label="Close bag" onClick={() => { setDrawer(false); setCheckout(false); setOrderPlaced(false) }}><X /></button></div>
          {orderPlaced ? <div className="confirmation"><span className="confirmation-mark"><Check /></span><h3>Thanks for stopping by.</h3><p>This is a storefront demo, so no order or payment was processed. Your interest means a lot.</p><a href="https://www.instagram.com/lad_n_lass_cloth" target="_blank" rel="noreferrer" className="dark-button">Find us on Instagram <ArrowUpRight size={15} /></a></div> : checkout ? <form className="checkout-form" onSubmit={(event) => { event.preventDefault(); setOrderPlaced(true); setCart([]) }}><p className="checkout-note">Just a demo checkout — no payment will be collected.</p><label>Full name<input required placeholder="Your name" /></label><label>Email address<input type="email" required placeholder="you@example.com" /></label><label>Phone number<input type="tel" required placeholder="+254" /></label><label>Delivery address<input required placeholder="Street, area, city" /></label><div className="checkout-total"><span>Estimated total</span><strong>{money(subtotal)}</strong></div><button type="submit" className="dark-button full-button">Place demo order <ArrowRight size={16} /></button><button type="button" className="back-link" onClick={() => setCheckout(false)}>Back to your bag</button></form> : <>{cart.length === 0 ? <div className="empty-cart"><ShoppingBag size={30} strokeWidth={1.25} /><h3>A little room for something good.</h3><p>Your bag is empty for now.</p><button className="dark-button" onClick={() => { setDrawer(false); document.getElementById("shop")?.scrollIntoView({ behavior: "smooth" }) }}>Explore the collection <ArrowRight size={16} /></button></div> : <><div className="cart-list">{cart.map(({ product, quantity }) => <div className="cart-item" key={product.id}><div className="cart-thumb"><Image src={product.image} alt={product.name} fill sizes="80px" className="product-image" /></div><div className="cart-item-details"><h3>{product.name}</h3><span>{money(product.price)}</span><div className="quantity-control"><button aria-label={`Decrease ${product.name} quantity`} onClick={() => updateQuantity(product.id, -1)}><Minus size={13} /></button><span>{quantity}</span><button aria-label={`Increase ${product.name} quantity`} onClick={() => updateQuantity(product.id, 1)}><Plus size={13} /></button></div></div><button className="remove-item" onClick={() => setCart((current) => current.filter((item) => item.product.id !== product.id))}>Remove</button></div>)}</div><div className="drawer-footer"><div className="subtotal"><span>Subtotal</span><strong>{money(subtotal)}</strong></div><p>Delivery calculated when you check out.</p><button className="dark-button full-button" onClick={() => setCheckout(true)}>Continue to checkout <ArrowRight size={16} /></button><button className="back-link" onClick={() => setDrawer(false)}>Keep looking</button></div></>}</>}
        </aside>}
      </div>}
    </main>
  )
}
